<script>
(function() {
  'use strict';

  const metals = [
    { id: 'magnesium',     name: 'Magnesium',         potential: -1.60 },
    { id: 'zinc',          name: 'Zinc / galvanized', potential: -1.05 },
    { id: 'aluminum',      name: 'Aluminum',          potential: -0.90 },
    { id: 'mild-steel',    name: 'Mild steel',        potential: -0.68 },
    { id: 'weather-steel', name: 'Weathering steel',  potential: -0.62 },
    { id: 'lead',          name: 'Lead',              potential: -0.52 },
    { id: 'brass',         name: 'Brass',             potential: -0.34 },
    { id: 'copper',        name: 'Copper',            potential: -0.28 },
    { id: 'bronze',        name: 'Bronze',            potential: -0.22 },
    { id: 'stainless',     name: 'Stainless steel',   potential: -0.05 },
    { id: 'titanium',      name: 'Titanium',          potential:  0.05 }
  ];

  const ENVIRONMENTS = {
    sheltered: { mult: 2.0, phrase: 'in sheltered exposure' },
    standard:  { mult: 1.0, phrase: 'in standard rain exposure' },
    coastal:   { mult: 0.5, phrase: 'in coastal exposure' }
  };

  const BASE = [0.15, 0.30, 0.50];

  const SEVERITIES = [
    { label: 'Compatible', token: 'compat'   },
    { label: 'Mild',       token: 'mild'     },
    { label: 'Moderate',   token: 'moderate' },
    { label: 'Severe',     token: 'severe'   }
  ];

  function descFor(sevIdx, envKey) {
    const p = ENVIRONMENTS[envKey].phrase;
    if (sevIdx === 0) return 'Negligible galvanic activity ' + p + '. Safe for direct contact in typical detailing.';
    if (sevIdx === 1) return envKey === 'coastal'
      ? 'Slow staining and pitting are likely over years of salt exposure. Watch joints that stay wet.'
      : 'Limited corrosion ' + p + '. Standard joint detailing handles it.';
    if (sevIdx === 2) return 'Notable corrosion of the more active metal whenever the joint stays wet ' + p + '. Detail to drain and isolate.';
    return envKey === 'coastal'
      ? 'Aggressive corrosion. This pairing should be avoided or fully isolated for any coastal application.'
      : 'Aggressive galvanic corrosion ' + p + '. Isolate fully or substitute one of the metals.';
  }

  function detailingText(sevIdx, envKey, anode) {
    if (sevIdx === 0) return 'Standard installation is fine. No special precautions required for this combination.';
    if (sevIdx === 1) return envKey === 'coastal'
      ? 'Use EPDM or butyl gaskets at fastener heads. Caulk laps and seal terminations to limit dwell time of salt spray.'
      : 'Use EPDM or butyl gaskets at fastener heads. Detail joints to shed water rather than pool it.';
    if (sevIdx === 2) {
      const sizeRule = ' Keep the anode (' + anode.name.toLowerCase() + ') the larger surface — a small anode against a large cathode concentrates the attack and fails fast.';
      return 'Isolate with neoprene, EPDM, or butyl tape between the metals. Use coated fasteners or fasteners matched to the more noble metal. Slope every joint to drain.' + sizeRule;
    }
    const sizeRule = ' Critical: never let the more active metal (' + anode.name.toLowerCase() + ') be the smaller piece. A small anode against a large cathode corrodes through within a few wet seasons.';
    if (envKey === 'coastal') return 'Avoid direct contact entirely. Use full dielectric separators (EPDM, butyl tape, plastic isolators), specify 316 stainless fasteners, and prefer powder-coated or PVDF-finished substrates.' + sizeRule;
    return 'Avoid direct contact. Use dielectric separators (EPDM, butyl tape, plastic isolators) at every contact point. Substitute one metal where the design allows.' + sizeRule;
  }

  function severityIndex(diff, envKey) {
    const a = Math.abs(diff);
    const m = ENVIRONMENTS[envKey].mult;
    const t = BASE.map(x => x * m);
    if (a < t[0]) return 0;
    if (a < t[1]) return 1;
    if (a < t[2]) return 2;
    return 3;
  }

  function fmtV(p) { return (p >= 0 ? '+' : '') + p.toFixed(2); }
  function escapeHtml(s) { return s.replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

  /* DOM */
  const m1Sel    = document.getElementById('metal1');
  const m2Sel    = document.getElementById('metal2');
  const resEl    = document.getElementById('result');
  const infoEl   = document.getElementById('info');
  const scaleSvg = document.getElementById('scale');

  metals.forEach((m, i) => {
    const v = fmtV(m.potential);
    m1Sel.add(new Option(m.name + ' (' + v + ' V)', i));
    m2Sel.add(new Option(m.name + ' (' + v + ' V)', i));
  });

  /* State */
  const state = { a: 2, b: 9, env: 'standard' };

  /* URL params */
  function readUrl() {
    try {
      const p = new URLSearchParams(location.search);
      const aId = p.get('a'), bId = p.get('b'), env = p.get('env');
      if (aId) { const i = metals.findIndex(m => m.id === aId); if (i >= 0) state.a = i; }
      if (bId) { const i = metals.findIndex(m => m.id === bId); if (i >= 0) state.b = i; }
      if (env && ENVIRONMENTS[env]) state.env = env;
    } catch (e) { /* same as writeUrl: ignore in sandboxed contexts */ }
  }
  function writeUrl() {
    try {
      const p = new URLSearchParams();
      p.set('a', metals[state.a].id);
      p.set('b', metals[state.b].id);
      p.set('env', state.env);
      history.replaceState(null, '', '?' + p.toString());
    } catch (e) {
      /* Sandboxed iframes and some file:// contexts disallow history manipulation.
         Deep-linking won't work there, but the tool still functions normally. */
    }
  }

  readUrl();
  m1Sel.value = state.a;
  m2Sel.value = state.b;
  document.querySelectorAll('.env-btns button').forEach(btn => {
    btn.classList.toggle('selected', btn.dataset.env === state.env);
  });

  /* Scale */
  const POT_MIN = -1.70, POT_MAX = 0.15;
  const SCALE_TOP = 56, SCALE_BOTTOM = 544;
  const yFor = p => SCALE_TOP + ((p - POT_MIN) / (POT_MAX - POT_MIN)) * (SCALE_BOTTOM - SCALE_TOP);

  function renderScale() {
    const m1 = metals[state.a], m2 = metals[state.b];
    const sev = SEVERITIES[severityIndex(m1.potential - m2.potential, state.env)];
    const sevC  = 'var(--c-' + sev.token + ')';
    const sevBg = 'var(--c-' + sev.token + '-bg)';

    const headStyle = 'font-family:var(--font-mono);font-size:11px;font-weight:500;letter-spacing:1.4px;text-transform:uppercase;fill:var(--ink);';
    const subStyle  = 'font-family:var(--font-mono);font-size:10px;letter-spacing:1.2px;text-transform:uppercase;fill:var(--ink-3);';

    let g = '';
    g += '<text x="120" y="22"  text-anchor="middle" style="' + headStyle + '">Active</text>';
    g += '<text x="120" y="38"  text-anchor="middle" style="' + subStyle  + '">corrodes</text>';
    g += '<text x="120" y="568" text-anchor="middle" style="' + headStyle + '">Noble</text>';
    g += '<text x="120" y="584" text-anchor="middle" style="' + subStyle  + '">protected</text>';

    g += '<line x1="120" y1="44" x2="120" y2="556" style="stroke:var(--rule);stroke-width:1;"/>';
    g += '<polygon points="115,47 125,47 120,40"   style="fill:var(--ink-3);"/>';
    g += '<polygon points="115,553 125,553 120,560" style="fill:var(--ink-3);"/>';

    metals.forEach((m, i) => {
      const y = yFor(m.potential);
      const sel = i === state.a || i === state.b;
      const tickC  = sel ? sevC : 'var(--ink-3)';
      const labelC = sel ? 'var(--ink)' : 'var(--ink-2)';
      const w  = sel ? '500' : '400';
      const sw = sel ? 2 : 1;
      g += '<line x1="115" y1="' + y + '" x2="125" y2="' + y + '" style="stroke:' + tickC + ';stroke-width:' + sw + ';"/>';
      g += '<text x="110" y="' + (y + 4) + '" text-anchor="end" style="font-family:var(--font-body);font-size:12px;font-weight:' + w + ';fill:' + labelC + ';">' + escapeHtml(m.name) + '</text>';
      g += '<text x="130" y="' + (y + 4) + '" text-anchor="start" style="font-family:var(--font-mono);font-size:11px;fill:' + labelC + ';">' + fmtV(m.potential) + '</text>';
    });

    [m1, m2].forEach(m => {
      const y = yFor(m.potential);
      g += '<circle cx="120" cy="' + y + '" r="6" style="fill:' + sevC + ';stroke:var(--bg-card);stroke-width:2;"/>';
    });

    if (state.a !== state.b) {
      const y1 = yFor(m1.potential), y2 = yFor(m2.potential);
      const yt = Math.min(y1, y2), yb = Math.max(y1, y2);
      const bx = 195;
      g += '<line x1="' + bx + '" y1="' + yt + '" x2="' + bx + '" y2="' + yb + '" style="stroke:' + sevC + ';stroke-width:2;"/>';
      g += '<line x1="' + (bx-4) + '" y1="' + yt + '" x2="' + (bx+4) + '" y2="' + yt + '" style="stroke:' + sevC + ';stroke-width:2;"/>';
      g += '<line x1="' + (bx-4) + '" y1="' + yb + '" x2="' + (bx+4) + '" y2="' + yb + '" style="stroke:' + sevC + ';stroke-width:2;"/>';
      const ym = (yt + yb) / 2;
      const dmv = Math.round(Math.abs(m1.potential - m2.potential) * 1000);
      g += '<rect x="' + (bx+8) + '" y="' + (ym-12) + '" width="68" height="24" rx="3" style="fill:' + sevBg + ';"/>';
      g += '<text x="' + (bx+42) + '" y="' + (ym+4) + '" text-anchor="middle" style="font-family:var(--font-mono);font-size:11px;font-weight:500;fill:' + sevC + ';">' + dmv + ' mV</text>';
    }

    scaleSvg.innerHTML = g;
  }

  /* Update */
  function update() {
    const m1 = metals[state.a], m2 = metals[state.b];

    if (state.a === state.b) {
      resEl.innerHTML = '<p style="font-family:var(--font-display);font-style:italic;font-size:1.15rem;color:var(--ink-2);max-width:48ch;margin:0 auto;">Same metal selected — no galvanic reaction.</p>';
      infoEl.innerHTML = '<div class="info-card"><div class="card-meta"><span class="card-num">01</span><span class="card-tag" style="color:var(--ink-3);">Note</span></div><h3>Self-pairing</h3><p>Identical metals at a junction have no electrochemical potential difference, so no galvanic cell forms. Pick two different metals to see how they interact.</p></div>';
      renderScale();
      writeUrl();
      return;
    }

    const diff   = Math.abs(m1.potential - m2.potential);
    const sevIdx = severityIndex(diff, state.env);
    const sev    = SEVERITIES[sevIdx];
    const anode   = m1.potential < m2.potential ? m1 : m2;
    const cathode = m1.potential < m2.potential ? m2 : m1;
    const dmv = Math.round(diff * 1000);
    const sevBg = 'var(--c-' + sev.token + '-bg)';
    const sevC  = 'var(--c-' + sev.token + ')';

    resEl.innerHTML =
      '<span class="result-pill" style="background:' + sevBg + ';color:' + sevC + ';">' + sev.label + ' reaction</span>' +
      '<div class="result-num" style="color:' + sevC + ';">' + dmv + '<span class="unit">mV</span></div>' +
      '<p class="result-desc">' + descFor(sevIdx, state.env) + '</p>';

    infoEl.innerHTML =
      '<div class="info-card">' +
        '<div class="card-meta"><span class="card-num">01</span><span class="card-tag" style="color:var(--c-severe);">Anode — corrodes</span></div>' +
        '<h3>' + escapeHtml(anode.name) + '</h3>' +
        '<div class="metal-pot">' + fmtV(anode.potential) + ' V vs. SCE</div>' +
        '<p>The more active metal at this junction. It loses electrons and dissolves into rainwater or condensate, producing pitting, white salts, or rust streaks on the building face.</p>' +
      '</div>' +
      '<div class="info-card">' +
        '<div class="card-meta"><span class="card-num">02</span><span class="card-tag" style="color:var(--c-compat);">Cathode — protected</span></div>' +
        '<h3>' + escapeHtml(cathode.name) + '</h3>' +
        '<div class="metal-pot">' + fmtV(cathode.potential) + ' V vs. SCE</div>' +
        '<p>The more noble metal. It accepts electrons and stays intact — actually shielded from corrosion at the anode\u2019s expense, the way galvanizing protects the steel beneath it.</p>' +
      '</div>' +
      '<div class="info-card">' +
        '<div class="card-meta"><span class="card-num">03</span><span class="card-tag" style="color:var(--ink-2);">Detailing</span></div>' +
        '<h3>How to build the joint</h3>' +
        '<p>' + detailingText(sevIdx, state.env, anode) + '</p>' +
      '</div>';

    renderScale();
    writeUrl();
  }

  /* Events */
  m1Sel.addEventListener('change', () => { state.a = parseInt(m1Sel.value); update(); });
  m2Sel.addEventListener('change', () => { state.b = parseInt(m2Sel.value); update(); });
  document.querySelectorAll('.env-btns button').forEach(btn => {
    btn.addEventListener('click', () => {
      state.env = btn.dataset.env;
      document.querySelectorAll('.env-btns button').forEach(b => b.classList.toggle('selected', b === btn));
      update();
    });
  });

  update();
})();
